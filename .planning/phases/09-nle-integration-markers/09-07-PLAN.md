  <action>
Extend `src/cmds/export.nim` with NLE marker export:

1. Add imports at top:
   ```nim
   import ../exports/[markers, aaf]
   import ../render/scoreviz
   ```

2. Add NLE format enum and mapping:
   ```nim
   type
     NLEFormat = enum
       nleNone        # Not NLE export mode
       nleFCP7XML     # Adobe Premiere, DaVinci Resolve
       nleFCPXML      # Final Cut Pro X
       nleEDL         # DaVinci Resolve, generic
       nleAAF         # After Effects, Media Composer

   proc parseNLETarget(target: string): NLEFormat =
     ## Parse NLE name or format string to NLEFormat
     case target.toLowerAscii()
     of "premiere", "fcp7xml", "fcp7", "resolve-xml":
       return nleFCP7XML
     of "fcpx", "finalcut", "fcpxml", "fcp11":
       return nleFCPXML
     of "resolve", "edl", "resolve-edl":
       return nleEDL
     of "aftereffects", "ae", "aaf", "mediacomposer", "avid":
       return nleAAF
     else:
       return nleNone
   ```

3. Add CLI arguments in argument parsing section:
   ```nim
   var nleTarget: string = ""
   var nleFormat: NLEFormat = nleNone
   var includeScoreViz: bool = true
   var scoreVizMode: ScoreVizMode = svmBoth
   var noMarkers: bool = false
   var noGraph: bool = false
   var noText: bool = false
   ```

4. Add argument handling:
   ```nim
   of "--nle":
     expecting = "nle"
   of "--no-markers":
     noMarkers = true
   of "--no-graph":
     noGraph = true
     if scoreVizMode == svmBoth:
       scoreVizMode = svmText
     elif scoreVizMode == svmGraph:
       includeScoreViz = false
   of "--no-text":
     noText = true
     if scoreVizMode == svmBoth:
       scoreVizMode = svmGraph
     elif scoreVizMode == svmText:
       includeScoreViz = false
   ```

   And in expecting handler:
   ```nim
   of "nle":
     nleTarget = key
     nleFormat = parseNLETarget(key)
     if nleFormat == nleNone:
       error &"Unknown NLE target: {key}. Use: premiere, fcpx, resolve, aftereffects, or format names (fcp7xml, fcpxml, edl, aaf)"
     expecting = ""
   ```

5. Update help text to include NLE options:
   ```
   NLE Export:
     --nle TARGET          Export for NLE: premiere, fcpx, resolve, aftereffects
                           Or format: fcp7xml, fcpxml, edl, aaf
     --no-markers          Skip timeline markers (engagement/scene/speaker)
     --no-graph            Skip score graph overlay track
     --no-text             Skip score text overlay track
   ```

6. Add NLE export handling after boundary adjustment check, before analyze-only:
   ```nim
   # NLE export mode
   if nleFormat != nleNone:
     echo &"Exporting to {nleTarget} format..."

     # Generate markers from engagement data, scene boundaries, and speaker changes
     var markers: seq[Marker] = @[]

     if not noMarkers:
       # 1. Engagement markers: from top clips by engagement score
       # Source: clips seq from project.json loaded earlier, sorted by engagementScore
       var rankedClips = clips.sortedByIt(-it.engagementScore)
       for rank, clip in rankedClips[0 ..< min(10, rankedClips.len)]:
         markers.add(createEngagementMarker(
           clip.startMs,
           int(clip.engagementScore * 100),  # Convert 0-1 to 0-100
           rank + 1  # 1-indexed rank
         ))

       # 2. Scene boundary markers: from clip start points (transitions between clips)
       # Source: clips seq - each clip start represents a potential scene boundary
       for i, clip in clips:
         if i > 0:  # Skip first clip (no boundary before it)
           markers.add(createSceneMarker(clip.startMs))

       # 3. Speaker change markers: from diarization data if available
       # Source: Look for speaker_segments in project data or run diarization
       if project.hasSpeakerSegments():
         var prevSpeaker = -1
         for seg in project.speakerSegments:
           if seg.speakerId != prevSpeaker:
             let speakerName = project.getSpeakerName(seg.speakerId)
             markers.add(createSpeakerMarker(
               seg.startMs,
               seg.speakerId,
               speakerName
             ))
             prevSpeaker = seg.speakerId

     let outputExt = case nleFormat
       of nleFCP7XML: ".xml"
       of nleFCPXML: ".fcpxml"
       of nleEDL: ".edl"
       of nleAAF: ".aaf"
       else: ".xml"

     let outputFile = effectiveOutputDir / &"markers{outputExt}"

     if dryRun:
       echo &"[Dry run] Would create: {outputFile}"
       quit(0)

     case nleFormat
     of nleFCP7XML:
       writeMarkersFCP7(inputPath, markers, outputFile)
     of nleFCPXML:
       writeMarkersFCPXML(inputPath, markers, outputFile)
     of nleEDL:
       exportMarkersEDL(markers, outputFile, extractFilename(inputPath))
     of nleAAF:
       try:
         exportAAF(inputPath, markers, outputFile)
       except AAFExportError as e:
         echo &"AAF export failed: {e.msg}"
         echo "Falling back to FCP7 XML..."
         writeMarkersFCP7(inputPath, markers, outputFile.replace(".aaf", ".xml"))
     else:
       discard

     echo &"Created: {outputFile}"

     # Generate score visualization if requested
     if includeScoreViz and clips.len > 0:
       # Convert clips to engagement segments for visualization
       var segments: seq[EngagementSegment] = @[]
       for clip in clips:
         segments.add(EngagementSegment(
           startMs: clip.startMs,
           endMs: clip.endMs,
           score: clip.engagementScore * 100  # scoreviz expects 0-100
         ))

       # Setup visualization parameters
       var vizParams = defaultScoreVizParams()
       vizParams.mode = scoreVizMode

       # Render score visualization video
       let vizOutputPath = effectiveOutputDir / "score_overlay.mp4"
       renderScoreVisualization(inputPath, segments, vizOutputPath, vizParams)
       echo &"Created: {vizOutputPath}"

     quit(0)
   ```
