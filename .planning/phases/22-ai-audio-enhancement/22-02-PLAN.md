       ```nim
       import std/[osproc, os, strformat, strutils]

       proc isDeepFilterAvailable*(): bool =
         ## Detect DeepFilterNet via Python module import (CONTEXT.md locked decision).
         let (_, code) = execCmdEx("python3 -c \"import df\"")
         result = code == 0

       proc whichDeepFilter*(): string =
         ## OQ-1 resolution: try `deepFilter` PATH script first, then `python3 -m df.enhance`.
         ## Returns the command prefix to invoke; empty string if neither works.
         let (_, code1) = execCmdEx("deepFilter --help")
         if code1 == 0:
           return "deepFilter"
         let (_, code2) = execCmdEx("python3 -c \"import df.enhance\"")
         if code2 == 0:
           return "python3 -m df.enhance"
         return ""

       proc runDeepFilter*(inputWav, outDir: string
                          ): tuple[ok: bool, outputPath: string, err: string] =
           ## Run DeepFilterNet on a 48kHz WAV. Caller MUST resample first.
           ## Pitfall 7: output path = outDir / extractFilename(inputWav).
           createDir(outDir)
           let cmdPrefix = whichDeepFilter()
           if cmdPrefix == "":
             return (false, "",
               "DeepFilterNet not installed. Run: pip install deepfilternet")
           let cmd = &"{cmdPrefix} -o {quoteShell(outDir)} {quoteShell(inputWav)}"
           let (output, code) = execCmdEx(cmd)
           if code != 0:
             return (false, "", output)
           let outFile = outDir / extractFilename(inputWav)
           if not fileExists(outFile):
             return (false, "",
               &"DeepFilterNet finished (rc=0) but output {outFile} missing")
           return (true, outFile, "")
       ```
