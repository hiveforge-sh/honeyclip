                             extraHeaders: openArray[tuple[key, val: string]] = @[]
                            ): HttpResponse {.base.} =
         raise newException(CatchableError, "abstract")

       method postMultipart*(c: RealHttpClient, url: string,
                             fieldName: string, fileBytes: string,
                             filename: string,
