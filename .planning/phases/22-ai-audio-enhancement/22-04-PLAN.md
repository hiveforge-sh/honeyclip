    import os, shutil, subprocess, tempfile, unittest
    from pathlib import Path
    import av  # PyAV — declared in CLAUDE.md as the test dep

    REPO = Path(__file__).resolve().parent.parent
    HONEYCLIP = REPO / "honeyclip"  # built by `nimble make`
    FIXTURES = REPO / "tests" / "fixtures"

    def make_video_with_audio(path: Path, audio_src: Path):
        """Wrap an audio fixture in an mp4 by adding a black video stream."""
        ffmpeg = REPO / "build" / "ffmpeg"
        subprocess.run([
            str(ffmpeg), "-y",
            "-f", "lavfi", "-i", "color=c=black:s=320x240:d=3",
            "-i", str(audio_src),
            "-c:v", "libx264", "-c:a", "aac", "-shortest",
            str(path),
        ], check=True, capture_output=True)

    class EnhanceE2E(unittest.TestCase):
        def setUp(self):
            self.tmp = Path(tempfile.mkdtemp(prefix="hc_enhance_e2e_"))
            self.video = self.tmp / "noisy.mp4"
            make_video_with_audio(self.video, FIXTURES / "audio_noisy.wav")

        def tearDown(self):
            shutil.rmtree(self.tmp, ignore_errors=True)

        def test_help(self):
            r = subprocess.run([str(HONEYCLIP), "enhance", "--help"],
                               capture_output=True, text=True)
            self.assertEqual(r.returncode, 0)
            self.assertIn("usage: honeyclip enhance", r.stdout)

        def test_arg_error_unknown_flag(self):
            r = subprocess.run([str(HONEYCLIP), "enhance", "--bogus", str(self.video)],
                               capture_output=True, text=True)
            self.assertEqual(r.returncode, 2)

        def test_audio_only_local_path(self):
            """If DeepFilterNet present, --backend=local --audio-only writes WAV.
            If absent, --backend=elevenlabs path is exercised with a fake key against
            a recorded mock — but without HTTP interception in E2E, we skip this case
            when DFN is unavailable."""
            dfn = subprocess.run(["python3", "-c", "import df"], capture_output=True)
            if dfn.returncode != 0:
                self.skipTest("DeepFilterNet not installed; manual verification covers this")
            out = self.tmp / "enhanced.wav"
            r = subprocess.run([
                str(HONEYCLIP), "enhance",
                "--backend=local", "--audio-only",
                "-o", str(out), str(self.video),
            ], capture_output=True, text=True)
            self.assertEqual(r.returncode, 0, msg=r.stderr)
            self.assertTrue(out.exists())
            with av.open(str(out)) as c:
                a = c.streams.audio[0]
                # Source was 48k stereo (audio_noisy.wav); output should preserve
                self.assertEqual(a.codec_context.sample_rate, 48000)

        def test_video_default_output_path(self):
            dfn = subprocess.run(["python3", "-c", "import df"], capture_output=True)
            if dfn.returncode != 0:
                self.skipTest("DeepFilterNet not installed")
            r = subprocess.run([
                str(HONEYCLIP), "enhance",
                "--backend=local", str(self.video),
            ], capture_output=True, text=True)
            self.assertEqual(r.returncode, 0, msg=r.stderr)
            expected = self.tmp / "noisy.mp4.enhanced.mp4"
            self.assertTrue(expected.exists())
            with av.open(str(expected)) as c:
                # video stream preserved
                self.assertGreaterEqual(len(c.streams.video), 1)
                # audio stream replaced
                self.assertGreaterEqual(len(c.streams.audio), 1)

        def test_cost_gate_non_tty_no_yes_denies(self):
            """Without --yes on non-TTY (subprocess captured stdin = non-TTY) and
            with backend=elevenlabs and a long-enough fake duration, the cost gate
            must refuse rather than spend money. We assert exit code 1 and no
            output file."""
            os.environ["ELEVENLABS_API_KEY"] = "test-key-not-real"
            try:

                subprocess.run([str(ffmpeg), "-y", "-f", "lavfi",
                                "-i", "anullsrc=r=48000:cl=mono:duration=660",
                                str(long_audio)], check=True, capture_output=True)
                long_video = self.tmp / "long.mp4"
                make_video_with_audio(long_video, long_audio)
                r = subprocess.run([
                    str(HONEYCLIP), "enhance",
                    "--backend=elevenlabs", str(long_video),

                # Expect exit 1 (cost denied) — NOT 0 (would mean we charged money!)
                self.assertNotEqual(r.returncode, 0)
                # No enhanced output exists
                self.assertFalse((self.tmp / "long.mp4.enhanced.mp4").exists())
            finally:
                os.environ.pop("ELEVENLABS_API_KEY", None)

        def test_no_backend_available_clean_error(self):
            """No DFN, no API key → exit 1 with both remediations in stderr."""
            os.environ.pop("ELEVENLABS_API_KEY", None)

            r = subprocess.run([
                str(HONEYCLIP), "enhance", str(self.video),

            self.assertNotEqual(r.returncode, 0)

            self.assertIn("deepfilternet", stderr_full.lower())
            self.assertIn("elevenlabs_api_key", stderr_full.lower())

    ```markdown
    ## Audio Enhancement

    Clean up dialogue audio with a 5-stage AI-powered enhancement chain:
    denoise → voice-isolation → de-reverb → de-ess → hum removal → loudness normalization.

    ```bash
    honeyclip enhance input.mp4
    ```

    By default, honeyclip auto-detects which stages your audio needs (hum at
    50/60 Hz, sibilance, noise floor) and runs only those, plus loudness
    normalization to -16 LUFS.

    ### Backends

    honeyclip supports two enhancement backends:

    - **Local (default when available):** [DeepFilterNet](https://github.com/Rikorose/DeepFilterNet)
      via Python subprocess. Install once:
      ```bash
      pip install deepfilternet
      ```
      Covers denoise + de-reverb. Free, runs entirely on your machine.

    - **API (auto-fallback or forced):** ElevenLabs Voice Isolator API. Set:
      ```bash
      export ELEVENLABS_API_KEY=sk_...
      ```
      or add to `~/.config/honeyclip/config.toml` (Linux/macOS) /
      `%APPDATA%\honeyclip\config.toml` (Windows):
      ```toml
      [elevenlabs]
      apiKey = "sk_..."
      ```
      Covers voice isolation + denoise. Approximate cost: **~$0.10/minute** of
      audio. honeyclip auto-allows runs ≤ $1.00, prompts you above $1.00, and
      refuses to spend on non-interactive runs unless you pass `--yes`.

    ### Backend selection

    ```bash
    honeyclip enhance input.mp4 --backend=auto         # default: local first, ElevenLabs fallback
    honeyclip enhance input.mp4 --backend=local        # force DeepFilterNet
    honeyclip enhance input.mp4 --backend=elevenlabs   # force API
    ```

    ### Per-stage control

    Each stage accepts `on | off | auto` (default `auto`):

    ```bash
    honeyclip enhance input.mp4 --hum-removal=on --de-ess=off
    honeyclip enhance input.mp4 --denoise=auto --voice-isolation=on
    ```

    ### Audio-only output

    Skip the video remux and emit a cleaned WAV:

    ```bash
    honeyclip enhance input.mp4 --audio-only -o cleaned.wav
    ```

    ### Cost confirmation bypass

    On non-interactive runs (CI, scripts), pass `--yes` to skip the cost prompt:

    ```bash
    honeyclip enhance input.mp4 --backend=elevenlabs --yes
    ```

    ### How it works

    1. honeyclip probes your audio (sample rate, channels, duration).
    2. The auto-detect analyzer scans for hum (narrowband 50/60 Hz energy),
       sibilance (4 kHz+ band ratio), and noise floor (FFT-based estimate).
    3. The enhancement chain runs:
       - **Local backend:** resample to 48 kHz mono → DeepFilterNet (denoise + de-reverb)
         → FFmpeg hum-removal → de-ess → two-pass loudnorm → resample back.
       - **API backend:** resample to 48 kHz mono → ElevenLabs (voice isolation +
         denoise) → decode response → FFmpeg hum-removal → de-ess → two-pass
         loudnorm → resample back.
    4. The cleaned audio replaces the original in the output video (`-c:v copy`),
       or is written directly with `--audio-only`.

    ### Limitations

    - Voice isolation runs **only on the API backend**. DeepFilterNet does not
      separate vocals from music; the local path skips this stage.
    - DeepFilterNet requires Python 3 and ~10 seconds per minute of audio on CPU.
    - ElevenLabs accepts up to 1 hour or 500 MB per request. Longer files are
      automatically chunked at silence boundaries and reassembled with
      crossfades.
    - Windows: works with both backends provided you have Python + DeepFilterNet
      OR a valid ELEVENLABS_API_KEY. The native binary itself does not bundle
      Python.
    ```
