    ```nim
    proc buildHumDeessCmd*(inputPath, outputPath: string,
                           notch50Hz, notch60Hz, deEss: bool): string =
      var filters: seq[string] = @[]
      if notch50Hz: filters.add "equalizer=f=50:t=h:w=10:g=-40"
      if notch60Hz: filters.add "equalizer=f=60:t=h:w=10:g=-40"
      if deEss:     filters.add "deesser=i=0.5:m=0.5:f=0.5"
      if filters.len == 0: filters.add "anull"
      result = &"./build/ffmpeg -y -i {quoteShell(inputPath)} " &
               &"-af {quoteShell(filters.join(\",\"))} {quoteShell(outputPath)}"

    proc runLoudnormTwoPassSubprocess*(inputPath, outputPath: string): bool =
      # Pass 1: measure (parse JSON from stderr)
      let p1 = &"./build/ffmpeg -y -i {quoteShell(inputPath)} " &
               "-af loudnorm=I=-16:LRA=11:TP=-1.5:print_format=json -f null - 2>&1"
      let (out1, code1) = execCmdEx(p1)
      if code1 != 0: return false
      # Find the JSON block in stderr; parse measured_I/measured_LRA/measured_TP/measured_thresh/offset
      let jsonStart = out1.rfind('{')
      let jsonEnd   = out1.rfind('}')
      if jsonStart < 0 or jsonEnd <= jsonStart: return false
      let measured = parseJson(out1[jsonStart .. jsonEnd])
      let mi  = measured["input_i"].getStr
      let mlra = measured["input_lra"].getStr
      let mtp = measured["input_tp"].getStr
      let mth = measured["input_thresh"].getStr
      let off = measured["target_offset"].getStr
      let p2filter = &"loudnorm=I=-16:LRA=11:TP=-1.5:linear=true:" &
                     &"measured_I={mi}:measured_LRA={mlra}:measured_TP={mtp}:" &
                     &"measured_thresh={mth}:offset={off}"
      let p2 = &"./build/ffmpeg -y -i {quoteShell(inputPath)} " &
               &"-af {quoteShell(p2filter)} {quoteShell(outputPath)}"
      let (_, code2) = execCmdEx(p2)
      result = code2 == 0

    proc decodeMp3ToWav*(inputPath, outputPath: string): bool =
      ## OQ-3 resolution: ElevenLabs response is likely MP3; probe & decode to WAV.
      let cmd = &"./build/ffmpeg -y -i {quoteShell(inputPath)} -f wav {quoteShell(outputPath)}"
      let (_, code) = execCmdEx(cmd)
      result = code == 0

    proc resampleTo48kMono*(inputPath, outputPath: string): bool =
      let cmd = &"./build/ffmpeg -y -i {quoteShell(inputPath)} " &
                "-ar 48000 -ac 1 -f wav " & quoteShell(outputPath)
      let (_, code) = execCmdEx(cmd)
      result = code == 0

    proc resampleFromWav*(inputPath, outputPath: string;
                          sampleRate: int; channels: int): bool =
      let cmd = &"./build/ffmpeg -y -i {quoteShell(inputPath)} " &
                &"-ar {sampleRate} -ac {channels} {quoteShell(outputPath)}"
      let (_, code) = execCmdEx(cmd)
      result = code == 0
    ```

       proc reassembleWithCrossfade*(chunks: seq[string]; outputPath: string;
                                     crossfadeMs: int = 50): bool =
         if chunks.len == 0: return false
         if chunks.len == 1:
           copyFile(chunks[0], outputPath)
           return true
         # Build acrossfade chain: [0][1]acrossfade=d={crossfadeMs/1000}[a01];
         #                        [a01][2]acrossfade=...[a012]; ...
         var inputs = ""
         for c in chunks:
           inputs.add &"-i {quoteShell(c)} "

         for i in 0 ..< chunks.high:
