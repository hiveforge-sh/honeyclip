      cmakeArgs.add("-DCMAKE_RC_COMPILER=x86_64-w64-mingw32-windres")
      cmakeArgs.add("-DCMAKE_FIND_ROOT_PATH_MODE_PROGRAM=NEVER")
      cmakeArgs.add("-DCMAKE_FIND_ROOT_PATH_MODE_LIBRARY=ONLY")
      cmakeArgs.add("-DCMAKE_FIND_ROOT_PATH_MODE_INCLUDE=ONLY")

        CMAKE_C_COMPILER=x86_64-w64-mingw32-gcc-posix \

<task type="auto">
  <name>Task 3: Add ccache handling for cross-compilation</name>
  <files>ae.nimble</files>
  <action>
    Add ccache configuration to prevent cache collisions between native and cross builds.

    Following research pitfall #7 (ccache false cache hits):

    1. Set separate ccache directory for cross-compilation:
    ```nim
    proc setupCcacheDir(crossWindows: bool) =
      let ccacheDir = if crossWindows:
        getEnv("HOME") / ".ccache/x86_64-w64-mingw32"
      else:
        getEnv("HOME") / ".ccache/native"
      putEnv("CCACHE_DIR", ccacheDir)
    ```

    2. Call setupCcacheDir at start of makeml/makemlwin tasks

    3. Add ccache to CMake if available:
    ```nim
    proc addCcacheIfAvailable(cmakeArgs: var seq[string]) =
      let (_, code) = gorgeEx("command -v ccache")
      if code == 0:
        cmakeArgs.add("-DCMAKE_C_COMPILER_LAUNCHER=ccache")
        cmakeArgs.add("-DCMAKE_CXX_COMPILER_LAUNCHER=ccache")
    ```

    4. Add to both cmakeBuild and onnxBuild procs

    5. Print ccache stats at end of build:
    ```nim
    proc printCcacheStats() =
      let (output, code) = gorgeEx("ccache -s")
      if code == 0:
        echo "ccache stats:"
        echo output
    ```
  </action>
  <verify>
    - Separate cache dirs: grep for "CCACHE_DIR" in ae.nimble
    - ccache launcher: grep for "CMAKE_C_COMPILER_LAUNCHER" in ae.nimble
  </verify>
  <done>
    ccache is configured with separate directories for native and cross builds to prevent false cache hits.
  </done>
