  <action>
Create `src/metadata/parser.nim` with:

1. **Template loading**:
   ```nim
   proc loadTemplate*(path: string): MetadataTemplate =
     let json = parseFile(path)
     result.version = json{"version"}.getInt(1)

     # Parse global metadata
     if "global" in json:
       for key, val in json["global"].pairs:
         result.global[key] = val.getStr()

     # Parse video stream metadata
     if "video" in json:
       for key, val in json["video"].pairs:
         result.video[key] = val.getStr()

     # Parse audio stream metadata
     if "audio" in json:
       for key, val in json["audio"].pairs:
         result.audio[key] = val.getStr()

     # Parse chapters
     if "chapters" in json:
       for chap in json["chapters"].items:
         result.chapters.add ChapterMarker(
           startMs: chap["start_ms"].getBiggestInt(),
           endMs: chap["end_ms"].getBiggestInt(),
           title: chap["title"].getStr()
         )
   ```

2. **Variable substitution** with these variables:
   - `${VIDEO_TITLE}` - from input filename without extension
   - `${AUTHOR_NAME}` - from environment or parameter
   - `${YEAR}` - current year (4 digits)
   - `${ISO_DATE}` - current date in ISO 8601 format
   - `${FILENAME}` - input filename with extension

   ```nim
   proc substituteVariables*(template: MetadataTemplate,
                             videoPath: string,
                             authorName: string = ""): MetadataTemplate =
     let now = now()
     let filename = extractFilename(videoPath)
     let title = filename.changeFileExt("")
     let author = if authorName != "": authorName
                  else: getEnv("HONEYCLIP_AUTHOR", "")

     proc substitute(s: string): string =
       result = s
       result = result.replace("${VIDEO_TITLE}", title)
       result = result.replace("${AUTHOR_NAME}", author)
       result = result.replace("${YEAR}", $now.year)
       result = result.replace("${ISO_DATE}", now.format("yyyy-MM-dd"))
       result = result.replace("${FILENAME}", filename)

     result = template
     for key in result.global.keys:
       result.global[key] = substitute(result.global[key])
     for key in result.video.keys:
       result.video[key] = substitute(result.video[key])
     for key in result.audio.keys:
       result.audio[key] = substitute(result.audio[key])
     for i in 0..<result.chapters.len:
       result.chapters[i].title = substitute(result.chapters[i].title)
   ```

3. **Template discovery** (look for .honeyclip-meta.json):
   ```nim
   proc findTemplate*(videoPath: string): string =
     # Check same directory as video
     let dir = parentDir(videoPath)
     let local = dir / ".honeyclip-meta.json"
     if fileExists(local):
       return local

     # Check home directory
     let home = getHomeDir() / ".honeyclip-meta.json"
     if fileExists(home):
       return home

     return ""  # No template found
   ```

Create `src/metadata/apply.nim` with:

1. **Metadata value escaping** (for ffmetadata format):
   ```nim
   proc escapeMetadataValue*(s: string): string =
     ## Escape special characters for ffmetadata INI format
     ## Characters: =, ;, #, \, newline need backslash escaping
     result = ""
     for c in s:
       if c in {'=', ';', '#', '\\'}:
         result.add '\\'
       if c == '\n':
         result.add "\\n"
       else:
         result.add c
   ```

2. **FFmetadata file generation**:
   ```nim
   proc generateFFMetadata*(template: MetadataTemplate): string =
     ## Generate ffmetadata file content for FFmpeg -i metadata.txt
     result = ";FFMETADATA1\n"

     # Global metadata
     for key, val in template.global:
       result &= &"{key}={escapeMetadataValue(val)}\n"

     # Chapter markers
     for chapter in template.chapters:
       result &= "\n[CHAPTER]\n"
       result &= "TIMEBASE=1/1000\n"
       result &= &"START={chapter.startMs}\n"
       result &= &"END={chapter.endMs}\n"
       result &= &"title={escapeMetadataValue(chapter.title)}\n"
   ```

3. **Write ffmetadata to temp file**:
   ```nim
   proc writeFFMetadataFile*(template: MetadataTemplate,
                             outputPath: string = ""): string =
     ## Write ffmetadata to file, return path
     ## If outputPath empty, use temp file
     let content = generateFFMetadata(template)
     let path = if outputPath != "": outputPath
                else: getTempDir() / "honeyclip_metadata.txt"
     writeFile(path, content)
     return path
   ```

4. **Convert template to AVDictionary**:
   ```nim
   proc templateToAvdict*(dst: ptr ptr AVDictionary,
                          metadata: Table[string, string]) =
     ## Apply metadata table to FFmpeg AVDictionary
     ## Uses existing dictToAvdict from util/dict
     dictToAvdict(dst, metadata)
   ```

5. **Apply metadata to format context** (for direct API use):
   ```nim
   proc applyToFormatContext*(ctx: ptr AVFormatContext,
                               template: MetadataTemplate) =
     ## Apply global metadata to format context
     templateToAvdict(addr ctx.metadata, template.global)

   proc applyToStream*(stream: ptr AVStream,
                       metadata: Table[string, string]) =
     ## Apply metadata to specific stream
     templateToAvdict(addr stream.metadata, metadata)
   ```

Import std/[tables, strformat, os], types, ../ffmpeg, ../util/dict.
