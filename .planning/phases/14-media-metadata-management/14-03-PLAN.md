
3. **Add argument parsing** in the case statement:
   ```nim
   of "--meta-template", "--meta":
     expecting = "meta-template"
   of "--meta-title":
     expecting = "meta-title"
   of "--meta-author":
     expecting = "meta-author"
   of "--meta-copyright":
     expecting = "meta-copyright"
   ```

   And in the expecting handler:

   ```nim
   of "meta-author": metaAuthorOverride = key
   of "meta-copyright": metaCopyrightOverride = key
   ```

4. **Add help text** in the help output:
   ```
   Metadata:
     --meta-template PATH    Apply metadata template during export
     --meta-title TEXT       Override title in template
     --meta-author TEXT      Override author in template
     --meta-copyright TEXT   Override copyright in template
   ```

5. **Load and process template** before export loop (after input validation):
   ```nim
   var metadataTemplate: MetadataTemplate
   var hasMetadata = false

   if metaTemplatePath != "":
     if not fileExists(metaTemplatePath):
       error &"Metadata template not found: {metaTemplatePath}"
     metadataTemplate = loadTemplate(metaTemplatePath)
     metadataTemplate = substituteVariables(metadataTemplate, inputPath, metaAuthorOverride)

     # Apply overrides
     var overrides = initTable[string, string]()
     if metaTitleOverride != "":
       overrides["title"] = metaTitleOverride
     if metaAuthorOverride != "":
       overrides["artist"] = metaAuthorOverride
     if metaCopyrightOverride != "":
       overrides["copyright"] = metaCopyrightOverride

     if overrides.len > 0:
       metadataTemplate = merge(metadataTemplate, overrides)

     hasMetadata = true
     info &"Loaded metadata template: {metaTemplatePath}"
   ```

6. **Pass metadata to export function** - modify the parallel export call to include metadata:
   ```nim
   # When calling the export/render functions, pass the metadata
   # This will require modifying the export function signatures
   # For now, generate ffmetadata file and pass path
   var metadataFilePath = ""
   if hasMetadata:
     metadataFilePath = writeFFMetadataFile(metadataTemplate)
   ```

7. **Modify FFmpeg subprocess calls** where clips are rendered to include metadata:
   ```nim
   # In the FFmpeg args building section, add:
   if metadataFilePath != "":
     ffmpegArgs.add("-i")
     ffmpegArgs.add(metadataFilePath)
     ffmpegArgs.add("-map_metadata")
     ffmpegArgs.add("1")
   ```

8. **Cleanup temp metadata file** at end of export:
   ```nim
   if metadataFilePath != "" and fileExists(metadataFilePath):
     removeFile(metadataFilePath)
   ```

Note: The exact integration depends on how exportcmd.nim currently calls FFmpeg. Look at the existing subprocess or av module calls and add metadata input there.
