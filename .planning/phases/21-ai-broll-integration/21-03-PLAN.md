#!/usr/bin/env python3
"""Phase 21 end-to-end tests for the `honeyclip broll` subcommand.

Run as part of: python3 tests/test.py

Requires: pip install av
"""

import json
import os
import subprocess
import sys
import tempfile
import unittest
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
FIXTURES = ROOT / "tests" / "fixtures"
RESOURCES = ROOT / "resources"


def find_binary():
    candidates = [ROOT / "honeyclip", ROOT / "honeyclip.exe",
                  ROOT / "build" / "honeyclip", ROOT / "build" / "honeyclip.exe"]
    for c in candidates:
        if c.exists():
            return str(c)
    raise FileNotFoundError(
        f"honeyclip binary not found in: {[str(c) for c in candidates]}. "
        f"Run `nimble make` first.")


def find_test_video():
    # Match what tests/test.py uses; fall back to any small mp4 in resources/.
    for ext in ("*.mp4", "*.mov"):
        for p in sorted(RESOURCES.glob(ext)):
            return str(p)
    raise FileNotFoundError(f"no test video in {RESOURCES}")


class BrollHelpTest(unittest.TestCase):
    def test_help_prints_usage_and_all_flags(self):
        binary = find_binary()
        r = subprocess.run([binary, "broll", "--help"],
                           capture_output=True, text=True, timeout=15)
        self.assertEqual(r.returncode, 0, msg=f"stderr: {r.stderr}")
        out = r.stdout
        self.assertIn("usage: honeyclip broll", out)
        for flag in ("--max-broll", "--min-spacing", "--backend",
                     "--comfy-url", "--workflow", "--render", "--yes"):
            self.assertIn(flag, out, msg=f"missing flag: {flag}")


class BrollOfflineFailureTest(unittest.TestCase):
    """ComfyUI unreachable + no GEMINI_API_KEY → graceful failure."""

    def test_unreachable_comfyui_no_key_errors_clearly(self):
        binary = find_binary()
        video = find_test_video()
        env = os.environ.copy()
        env.pop("GEMINI_API_KEY", None)
        with tempfile.TemporaryDirectory() as td:
            out_path = Path(td) / "broll.json"
            r = subprocess.run(
                [binary, "broll", video,
                 "--backend", "auto",
                 "--comfy-url", "http://127.0.0.1:1",  # closed port
                 "--workflow", str(FIXTURES / "comfyui_workflow.json"),
                 "-o", str(out_path),
                 "--quiet"],
                capture_output=True, text=True, env=env, timeout=120)
            self.assertNotEqual(r.returncode, 0,
                                msg="expected non-zero on no-backend; got 0")
            combined = (r.stdout + r.stderr).lower()
            self.assertTrue(
                "comfyui" in combined or "gemini" in combined or "backend" in combined,
                msg=f"error message must reference the backend issue: {combined}")


class BrollMarkerSnapshotTest(unittest.TestCase):
    """Lock down the marker JSON shape against tests/fixtures/broll_marker_snapshot.json."""

    REQUIRED_TOP_KEYS = {"sourceVideo", "insertions"}
    REQUIRED_INSERT_KEYS = {"startMs", "endMs", "pauseMs", "score", "sourceTag",
                             "contextText", "prompt", "clipPath",
                             "actualDurationMs", "success"}

    def test_marker_json_matches_snapshot_shape(self):
        snapshot = json.loads((FIXTURES / "broll_marker_snapshot.json").read_text())
        # Snapshot file IS the source of truth for shape; we just assert the
        # required keys are present at both levels.
        self.assertEqual(set(snapshot.keys()), self.REQUIRED_TOP_KEYS)
        for ins in snapshot["insertions"]:
            self.assertEqual(set(ins.keys()), self.REQUIRED_INSERT_KEYS)


class BrollAudioSyncSmokeTest(unittest.TestCase):
    """If --render runs successfully (requires real ComfyUI), source and output
    duration must match within 100ms. In CI this test SKIPs with explanation."""

    def test_render_preserves_duration(self):
        if os.environ.get("HONEYCLIP_E2E_LIVE") != "1":
            self.skipTest("HONEYCLIP_E2E_LIVE != 1 (live render requires ComfyUI/Veo)")
        try:
            import av  # noqa: F401
        except ImportError:
            self.skipTest("PyAV (`pip install av`) not installed")
        # Live path implementation deferred to manual verification per VALIDATION.md.
        self.skipTest("live render covered by manual verifications in VALIDATION.md")


if __name__ == "__main__":
    unittest.main(verbosity=2)
```

```markdown
## B-Roll

`honeyclip broll` auto-detects narrative-meaningful pause points in your video,
generates short B-roll clips for each (locally via ComfyUI or remotely via Gemini Veo),
and either emits a marker JSON for review or muxes a final video.

### Quick start (local — ComfyUI)

```bash
# Start a ComfyUI server (defaults to http://127.0.0.1:8188).
# Export your workflow as API-format JSON. Mark the prompt, seed, and frame-count
# inputs with placeholders {{PROMPT}}, {{SEED}}, {{LENGTH}}.

honeyclip broll input.mp4 --workflow my-workflow.json
# → writes input_broll.json with detected insertion points and generated clips

honeyclip broll input.mp4 --workflow my-workflow.json --render -o final.mp4
# → muxes final.mp4 with B-roll inserts; original audio plays underneath
```

### Quick start (remote — Gemini Veo)

```bash
export GEMINI_API_KEY="your-key-here"
honeyclip broll input.mp4 --backend veo --max-broll 3 --yes
```

You can also store the key in a config file:

- Linux/macOS: `$XDG_CONFIG_HOME/honeyclip/config.toml` (default `~/.config/honeyclip/config.toml`)
- Windows:     `%APPDATA%\honeyclip\config.toml`

```toml
[api]
geminiKey = "your-key-here"
comfyuiUrl = "http://127.0.0.1:8188"
```

After creating the file:

```bash
chmod 600 ~/.config/honeyclip/config.toml
```

Add `**/honeyclip/config.toml` to your global `.gitignore`. honeyclip will warn
if the resolved config path falls inside a git repo.

### Backend selection

| Flag value      | Behavior |
|-----------------|----------|
| `--backend auto`     | Tries ComfyUI first; falls back to Veo if a key is set |
| `--backend comfyui`  | Forces ComfyUI; requires `--workflow PATH` |
| `--backend veo`      | Forces Veo; requires `GEMINI_API_KEY` |

### Cost note (Veo)

Veo bills per second of generated video. Each clip is fixed at 8 seconds.

| Tier               | Per second | Per 8s clip |
|--------------------|------------|-------------|
| `veo-3.0-fast-generate-001` (default) | $0.15 | $1.20 |
| `veo-3.0-generate-001`                | $0.40 | $3.20 |

Worked example: a 30-minute video with default density (~1 clip per 60s) would request
30 clips × 8s × $0.15/s ≈ **$36**. Use `--max-broll N` to cap.

honeyclip prompts for cost confirmation when running Veo on a TTY. Use `--yes` for
non-interactive runs (CI, scripts).

### SSRF caveat

`--comfy-url` accepts any URL. Default is `http://127.0.0.1:8188`. Do not point it
at internal services unintentionally — honeyclip does not validate the host.

### Output formats

- Default: `input_broll.json` — list of detected insertion points + generated clip paths.
  Reviewable before rendering.
- With `--render`: `input_broll.mp4` — fully muxed output with B-roll inserts spliced
  in, original audio playing underneath. A `_broll.json` sidecar is also written.

### Limits

- Veo clips are always 8s; honeyclip caps requested pauses at 6s and truncates as needed.
- ComfyUI duration depends on your workflow. honeyclip falls back to freeze-pad or speed-ramp
  fitting if the model returns a shorter clip than requested.
- ComfyUI workflow placeholder system is opt-in: only `{{PROMPT}}`, `{{SEED}}`, `{{LENGTH}}`
  are substituted. Any other tokens pass through unchanged.

### Caveat: first network feature

honeyclip is otherwise offline-first. The `broll` subcommand is the only one that
talks to external services (ComfyUI on localhost or Gemini APIs). API keys are
read from env first, config file second, and never written to logs.
