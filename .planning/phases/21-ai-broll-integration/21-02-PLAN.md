```nim
import std/strutils

proc escapeJsonString*(s: string): string =
  ## Escape a string so it's safe to embed inside an existing JSON string literal.
  result = newStringOfCap(s.len + 8)
  for ch in s:
    case ch
    of '"': result.add "\\\""
    of '\\': result.add "\\\\"
    of '\n': result.add "\\n"
    of '\r': result.add "\\r"
    of '\t': result.add "\\t"
    of '\b': result.add "\\b"
    of '\f': result.add "\\f"
    elif ch.ord < 0x20:
      result.add "\\u00" & toHex(ch.ord, 2)
    else:
      result.add ch

proc substituteWorkflow*(workflowJson: string, prompt: string,
                          seed: int64, lengthFrames: int): string =
  result = workflowJson
    .replace("{{PROMPT}}", escapeJsonString(prompt))
    .replace("{{SEED}}", $seed)
    .replace("{{LENGTH}}", $lengthFrames)
```

```nim
import std/[json, os, random, strformat, strutils, times]
import ../http_client
import ../analyze/broll as brollAnalyze
import ../log
import workflow

type
  ClipResult* = object
    success*: bool
    clipPath*: string
    actualDurationMs*: int64
    error*: string

proc isComfyReachable*(client: HttpClientLike, baseUrl: string,
                       timeoutMs: int = 2_000): bool =
  try:
    let (code, _) = client.get(baseUrl & "/system_stats")
    result = code == 200
  except CatchableError:
    result = false

proc comfyuiGenerate*(client: HttpClientLike, baseUrl: string,
                     workflowPath: string,
                     insertion: brollAnalyze.BrollInsertion,
                     prompt: string, outDir: string,
                     pollIntervalMs: int = 1_000,
                     maxAttempts: int = 300): ClipResult =
  ## Submit, poll, download. Synchronous; ComfyUI poll is short.
  if not fileExists(workflowPath):
    return ClipResult(success: false,
                      error: &"workflow file not found: {workflowPath}")
  let workflowRaw = readFile(workflowPath)
  let lengthFrames = max(8, int(insertion.pauseMs * 24 div 1000))
  randomize()
  let seed = rand(int64.high div 2)
  let promptBody = substituteWorkflow(workflowRaw, prompt, seed, lengthFrames)
  let body = "{\"prompt\":" & promptBody & ",\"client_id\":\"honeyclip\"}"

  try:
    let (postCode, postResp) = client.post(baseUrl & "/prompt", body)
    if postCode notin 200 .. 299:
      return ClipResult(success: false,
                        error: &"ComfyUI /prompt returned {postCode}: {postResp}")
    let promptId = parseJson(postResp){"prompt_id"}.getStr
    if promptId == "":
      return ClipResult(success: false, error: "ComfyUI did not return prompt_id")

    for attempt in 0 ..< maxAttempts:
      sleep(pollIntervalMs)
      let (histCode, histResp) = client.get(baseUrl & "/history/" & promptId)
      if histCode != 200: continue
      let hist = parseJson(histResp)
      if not hist.hasKey(promptId): continue
      let outputs = hist[promptId]{"outputs"}
      if outputs == nil or outputs.kind != JObject: continue
      for nodeId, nodeOut in outputs.fields:
        for kind in ["videos", "gifs", "images"]:
          if nodeOut.hasKey(kind) and nodeOut[kind].len > 0:
            let f = nodeOut[kind][0]
            let filename = f{"filename"}.getStr
            let subfolder = f{"subfolder"}.getStr
            let viewUrl = &"{baseUrl}/view?filename={filename}&subfolder={subfolder}&type=output"
            let (dlCode, dlBody) = client.get(viewUrl)
            if dlCode != 200:
              return ClipResult(success: false,
                                error: &"ComfyUI /view returned {dlCode}")
            createDir(outDir)
            let outPath = outDir / &"broll_{insertion.startMs}.mp4"
            writeFile(outPath, dlBody)
            return ClipResult(success: true, clipPath: outPath,
                              actualDurationMs: insertion.pauseMs)
    return ClipResult(success: false,
                      error: &"ComfyUI timeout after {maxAttempts * pollIntervalMs} ms")
  except CatchableError as e:
    return ClipResult(success: false, error: e.msg)
```

```nim
import std/[json, os, strformat, strutils, times]
import ../http_client
import ../analyze/broll as brollAnalyze
import ../log
import comfyui  # for ClipResult; or move ClipResult into a shared types module

const
  veoBase* = "https://generativelanguage.googleapis.com/v1beta"
  veoFastSecPrice* = 0.15
  veoStandardSecPrice* = 0.40
  veoFixedClipSeconds* = 8

proc estimateVeoCost*(numClips: int, model: string): float =
  let perSec = if "fast" in model.toLowerAscii: veoFastSecPrice
               else: veoStandardSecPrice
  result = numClips.float * veoFixedClipSeconds.float * perSec

proc confirmVeoCost*(numClips: int, model: string,
                     isTty: bool, autoYes: bool,
                     readLineFn: proc(): string = nil): bool =
  if autoYes: return true
  let cost = estimateVeoCost(numClips, model)
  if not isTty:
    error &"Veo would generate {numClips} clips ≈ ${cost:.2f}; refusing without --yes on non-TTY"
    return false
  echo &"Veo will generate {numClips} clips × {veoFixedClipSeconds}s ≈ ${cost:.2f} USD. Continue? [y/N]"
  let line = if readLineFn != nil: readLineFn() else: stdin.readLine()
  return line.strip.toLowerAscii in ["y", "yes"]

proc veoGenerate*(client: HttpClientLike, apiKey: string, model: string,
                  insertion: brollAnalyze.BrollInsertion,
                  prompt: string, outDir: string,
                  pollIntervalMs: int = 10_000,
                  maxAttempts: int = 60): ClipResult =
  if apiKey == "":
    return ClipResult(success: false, error: "Veo requires GEMINI_API_KEY")
  try:
    let body = $(%*{
      "instances": [{"prompt": prompt}],
      "parameters": {"aspectRatio": "16:9", "personGeneration": "allow_adult"}
    })
    let submitUrl = &"{veoBase}/models/{model}:predictLongRunning?key={apiKey}"
    let (postCode, postResp) = client.post(submitUrl, body)
    if postCode notin 200 .. 299:
      return ClipResult(success: false, error: &"Veo submit returned {postCode}")
    let opName = parseJson(postResp){"name"}.getStr
    if opName == "":
      return ClipResult(success: false, error: "Veo did not return operation name")

    let pollUrl = &"{veoBase}/{opName}?key={apiKey}"
    for attempt in 0 ..< maxAttempts:
      sleep(pollIntervalMs)
      let (pollCode, pollResp) = client.get(pollUrl)
      if pollCode != 200: continue
      let op = parseJson(pollResp)
      if not (op.hasKey("done") and op["done"].getBool): continue
      if op.hasKey("error"):
        return ClipResult(success: false,
                          error: op["error"]{"message"}.getStr("Veo error"))
      let videoUri = op{"response"}{"generateVideoResponse"}{"generatedSamples"}[0]{"video"}{"uri"}.getStr
      if videoUri == "":
        return ClipResult(success: false, error: "Veo response missing video URI")
      let dlUrl = if "?" in videoUri: videoUri & "&key=" & apiKey
                   else: videoUri & "?key=" & apiKey
      let (dlCode, dlBody) = client.get(dlUrl)
      if dlCode != 200:
        return ClipResult(success: false, error: &"Veo download returned {dlCode}")
      createDir(outDir)
      let outPath = outDir / &"broll_veo_{insertion.startMs}.mp4"
      writeFile(outPath, dlBody)
      return ClipResult(success: true, clipPath: outPath,
                        actualDurationMs: int64(veoFixedClipSeconds * 1000))
    return ClipResult(success: false, error: "Veo timeout")
  except CatchableError as e:
    return ClipResult(success: false, error: e.msg)
```

```nim
import std/[json, strformat, strutils]
import ../http_client
import ../log

const flashModel* = "gemini-2.5-flash"
const flashBase* = "https://generativelanguage.googleapis.com/v1beta"

proc rewritePrompt*(client: HttpClientLike, apiKey: string,
                    contextText: string): string =
  ## Rewrite a transcript window into a concrete visual prompt.
  ## On any failure, returns contextText unchanged after logging a warning.
  if apiKey == "" or contextText == "":
    return contextText
  let instruction = "Rewrite this spoken transcript snippet into a concise visual scene description suitable for a text-to-video model. Output ONLY the scene description, no preamble. Snippet: " & contextText
  let body = $(%*{
    "contents": [{"parts": [{"text": instruction}]}]
  })
  try:
    let url = &"{flashBase}/models/{flashModel}:generateContent?key={apiKey}"
    let (code, resp) = client.post(url, body)
    if code != 200:
      warning &"Gemini Flash returned {code}; using raw transcript"
      return contextText
    let parsed = parseJson(resp)
    let txt = parsed{"candidates"}[0]{"content"}{"parts"}[0]{"text"}.getStr
    if txt == "": return contextText
    return txt.strip
  except CatchableError as e:
    warning &"Gemini Flash rewrite failed: {e.msg}; using raw transcript"
```

```nim
import std/[osproc, strformat, strutils]

type
  FittingStrategy* = enum
    fsNone           ## actual within tolerance of requested
    fsTruncate       ## actual > requested → trim to requested
    fsPadFreeze      ## requested > actual by ≤500ms → freeze last frame
    fsPadSpeedRamp   ## requested > actual by ≤2000ms → setpts speed ramp
    fsRequestExtra   ## gap too large for safe fitting; caller decides

const fittingToleranceMs* = 200
const padFreezeMaxGapMs* = 500
const padSpeedMaxGapMs*  = 2000

proc selectFittingStrategy*(actualMs, requestedMs: int64): FittingStrategy =
  let diff = actualMs - requestedMs
  if abs(diff) <= fittingToleranceMs: return fsNone
  if diff > 0: return fsTruncate
  let gap = -diff
  if gap <= padFreezeMaxGapMs: return fsPadFreeze
  if gap <= padSpeedMaxGapMs: return fsPadSpeedRamp
  return fsRequestExtra

proc buildFittingCmd*(clipPath: string, requestedMs, actualMs: int64,
                      strategy: FittingStrategy, outPath: string): string =
  ## Pure builder — separated for testability.
  case strategy
  of fsNone, fsRequestExtra: result = ""
  of fsTruncate:
    result = &"ffmpeg -y -i {clipPath.quoteShell} -t {requestedMs.float/1000:.3f} -c copy {outPath.quoteShell}"
  of fsPadFreeze:
    let padSec = (requestedMs - actualMs).float / 1000.0
    result = &"ffmpeg -y -i {clipPath.quoteShell} -vf tpad=stop_mode=clone:stop_duration={padSec:.3f} {outPath.quoteShell}"
  of fsPadSpeedRamp:
    let speedRatio = actualMs.float / requestedMs.float
    result = &"ffmpeg -y -i {clipPath.quoteShell} -vf setpts={1.0 / speedRatio:.4f}*PTS {outPath.quoteShell}"

proc fitClipDuration*(clipPath: string, requestedMs, actualMs: int64,
                      strategy: FittingStrategy, outPath: string): bool =
  let cmd = buildFittingCmd(clipPath, requestedMs, actualMs, strategy, outPath)
  if cmd == "": return false
  let (_, exitCode) = execCmdEx(cmd)
  return exitCode == 0
```

```nim
import std/[json, os, osproc, strformat, strutils]
import ../analyze/broll as brollAnalyze
import ../log
import comfyui  # for ClipResult

type
  SegmentKind* = enum skSource, skBroll
  SegmentPlan* = object
    kind*: SegmentKind
    startMs*: int64
    endMs*: int64
    sourceIndex*: int   # for skBroll: index into clips[]; -1 for skSource

proc buildSegmentPlan*(durationMs: int64,
                       insertions: seq[brollAnalyze.BrollInsertion]): seq[SegmentPlan] =
  ## Walk insertions in chronological order, emit alternating source/broll segments.
  ## Caller guarantees insertions are sorted and non-overlapping.
  var cursor: int64 = 0
  for i, ins in insertions:
    if ins.startMs > cursor:
      result.add SegmentPlan(kind: skSource, startMs: cursor, endMs: ins.startMs, sourceIndex: -1)
    result.add SegmentPlan(kind: skBroll, startMs: ins.startMs, endMs: ins.endMs, sourceIndex: i)
    cursor = ins.endMs
  if cursor < durationMs:
    result.add SegmentPlan(kind: skSource, startMs: cursor, endMs: durationMs, sourceIndex: -1)

proc buildConcatList*(plan: seq[SegmentPlan], srcSegmentPaths: seq[string],
                      brollClipPaths: seq[string]): string =
  ## srcSegmentPaths is parallel to occurrences of skSource in `plan`.
  var srcIdx = 0
  for s in plan:
    case s.kind
    of skSource:
      result.add &"file '{srcSegmentPaths[srcIdx]}'\n"
      inc srcIdx
    of skBroll:
      result.add &"file '{brollClipPaths[s.sourceIndex]}'\n"

proc normalizeBrollClipCmd*(brollPath: string, targetW, targetH: int,
                            targetFps: float, outPath: string): string =
  &"ffmpeg -y -i {brollPath.quoteShell} -vf scale={targetW}:{targetH},fps={targetFps:.3f} -c:v libx264 -preset fast -an {outPath.quoteShell}"

proc renderWithBroll*(sourceVideo: string,
                      insertions: seq[brollAnalyze.BrollInsertion],
                      clips: seq[ClipResult],
                      durationMs: int64,
                      outputPath: string,
                      tmpDir: string = ""): bool =
  let workDir = if tmpDir != "": tmpDir
                else: getTempDir() / &"honeyclip_broll_{epochTime():.0f}"
  createDir(workDir)
  try:
    # 1. Probe source for w/h/fps via ffprobe (or fall back to known FFmpeg invocation).
    #    For Phase 21 Wave 3, accept that ffprobe is on PATH (same assumption as Phase 19-02).
    let probeCmd = &"ffprobe -v error -select_streams v:0 -show_entries stream=width,height,r_frame_rate -of csv=p=0 {sourceVideo.quoteShell}"
    let (probeOut, probeCode) = execCmdEx(probeCmd)
    if probeCode != 0:
      error &"ffprobe failed on {sourceVideo}: {probeOut}"
      return false
    let parts = probeOut.strip.split(',')
    let targetW = parts[0].parseInt
    let targetH = parts[1].parseInt
    let fpsParts = parts[2].split('/')
    let targetFps = fpsParts[0].parseFloat / fpsParts[1].parseFloat

    # 2. Normalize each broll clip.
    var normalizedPaths: seq[string]
    for i, c in clips:
      if not c.success:
        warning &"Skipping failed clip at insertion {i}"
        return false
      let normPath = workDir / &"broll_norm_{i}.mp4"
      let cmd = normalizeBrollClipCmd(c.clipPath, targetW, targetH, targetFps, normPath)
      let (out, code) = execCmdEx(cmd)
      if code != 0:
        error &"Normalize failed: {out}"
        return false
      normalizedPaths.add normPath

    # 3. Build segment plan + extract source segments.
    let plan = buildSegmentPlan(durationMs, insertions)
    var srcSegmentPaths: seq[string]
    var srcCount = 0
    for s in plan:
      if s.kind == skSource:
        let segPath = workDir / &"src_seg_{srcCount}.mp4"
        let durSec = (s.endMs - s.startMs).float / 1000.0
        let startSec = s.startMs.float / 1000.0
        let cmd = &"ffmpeg -y -ss {startSec:.3f} -t {durSec:.3f} -i {sourceVideo.quoteShell} -c:v libx264 -preset fast -an {segPath.quoteShell}"
        let (out, code) = execCmdEx(cmd)
        if code != 0:
          error &"Source segment extraction failed: {out}"
          return false
        srcSegmentPaths.add segPath
        inc srcCount

    # 4. Write concat list and concat.
    let listPath = workDir / "concat_list.txt"
    writeFile(listPath, buildConcatList(plan, srcSegmentPaths, normalizedPaths))
    let tempVideo = workDir / "temp_video.mp4"
    let concatCmd = &"ffmpeg -y -f concat -safe 0 -i {listPath.quoteShell} -c copy {tempVideo.quoteShell}"
    let (concatOut, concatCode) = execCmdEx(concatCmd)
    if concatCode != 0:
      error &"Concat failed: {concatOut}"
      return false

    # 5. Extract source audio.
    let srcAudio = workDir / "source_audio.aac"
    let audioCmd = &"ffmpeg -y -i {sourceVideo.quoteShell} -vn -c:a aac {srcAudio.quoteShell}"
    let (audioOut, audioCode) = execCmdEx(audioCmd)
    if audioCode != 0:
      error &"Audio extraction failed: {audioOut}"
      return false

    # 6. Mux video + audio.
    let muxCmd = &"ffmpeg -y -i {tempVideo.quoteShell} -i {srcAudio.quoteShell} -c:v copy -c:a copy -map 0:v:0 -map 1:a:0 {outputPath.quoteShell}"
    let (muxOut, muxCode) = execCmdEx(muxCmd)
    if muxCode != 0:
      error &"Mux failed: {muxOut}"
      return false
    return true
  finally:
    try: removeDir(workDir) except CatchableError: discard
```

Pipeline (copy chapters.nim shape):

```nim
proc main*(cArgs: seq[string]) =
  var inputPath, model = ""
  var backend = "auto"
  var comfyUrl = ""             # resolved from config if empty
  var workflowPath = ""
  var veoModel = "veo-3.0-fast-generate-001"
  var maxBroll = -1
  var minSpacingSec = 30
  var doRender = false
  var outputPath = ""
  var autoYes = false
  var quiet = false
  var debug = false

  # ... arg parser state machine (mirror chapters.nim line-for-line) ...

  if inputPath == "":
    echo helpText; quit(1)

  # Resolve config
  let cfg = loadConfig()
  let apiKey = resolveGeminiKey()
  if comfyUrl == "":
    comfyUrl = resolveComfyUrl()

  let http: HttpClientLike = newRealHttpClient()

  # Resolve provider EARLY so we fail fast.
  let provider = resolveProvider(backend, comfyUrl, workflowPath, apiKey, veoModel, http)

  # Run transcript + engagement (mirror chapters.nim's pipeline).
  # ... reuse extractTranscript + analyze.engagement ...
  let transcript = ...   # from src/transcript/extract
  let engagement = ...   # from src/analyze/engagement

  var params = defaultBrollParams()
  params.minSpacingMs = int64(minSpacingSec) * 1000
  params.maxInsertions = maxBroll

  let candidates = detectBrollInsertions(transcript, engagement, params)
  let chosen = selectInsertions(candidates, params, engagement.duration)

  if chosen.len == 0:
    echo "No B-roll insertion points detected."
    quit(0)

  # Veo cost gate
  if provider.kind == bpVeo:
    let isTty = isatty(stdout)
    if not confirmVeoCost(chosen.len, provider.model, isTty, autoYes):
      error "Aborted by user (or non-TTY without --yes)."
      quit(2)

  let outDir = getTempDir() / &"honeyclip_broll_{epochTime():.0f}"
  createDir(outDir)
  var clips: seq[ClipResult]
  var prompts: seq[string]
  for ins in chosen:
    let visualPrompt = rewritePrompt(http, apiKey, ins.contextText)
    prompts.add visualPrompt
    var r = generate(provider, http, ins, visualPrompt, outDir)
    if r.success and abs(r.actualDurationMs - ins.pauseMs) > 200:
      let strat = selectFittingStrategy(r.actualDurationMs, ins.pauseMs)
      if strat notin {fsNone, fsRequestExtra}:
        let fitted = outDir / &"broll_fit_{ins.startMs}.mp4"
        if fitClipDuration(r.clipPath, ins.pauseMs, r.actualDurationMs, strat, fitted):
          r.clipPath = fitted
          r.actualDurationMs = ins.pauseMs
    clips.add r

  # Determine output path
  let (dir, name, ext) = splitFile(inputPath)
  let defaultMarker = dir / name & "_broll.json"
  let defaultRender = dir / name & "_broll" & ext
  let resolvedOut = if outputPath != "":
    outputPath
  elif doRender: defaultRender
  else: defaultMarker

  if doRender:
    let markerSidecar = changeFileExt(resolvedOut, "json")
    writeBrollMarkersJson(inputPath, chosen, clips, prompts, markerSidecar)
    if not renderWithBroll(inputPath, chosen, clips, engagement.duration, resolvedOut):
      error "Render failed; marker JSON written to " & markerSidecar
      quit(3)
    info &"Wrote {resolvedOut} and marker sidecar {markerSidecar}"
  else:
    writeBrollMarkersJson(inputPath, chosen, clips, prompts, resolvedOut)
    info &"Wrote {resolvedOut}"
```
